# AO's MCP does many at once (#108): add_cards, add_tasks, update_tasks with their order and
# remove_cards, each item as the tool for one takes it. list_tasks keeps to a milestone and a status
# with only the fields asked for, a task can be given evidence (a picture or a file), and a milestone
# has notes. Run bulk-tasks-again.txt after it, with the same data folder, to see them kept.
# 6 tasks under 2 milestones: Tasks 1, 3 and 5 in Milestone 1, Tasks 2, 4 and 6 in Milestone 2; 1 and 4 done.
seed "Bulk project" 6 2 0
# Many cards in one call. One already there is refused, saying why, and the rest are added.
mcpjson "Bulk project" add_cards "{\qcards\q:[{\qpage\q:\qAssets\q,\qname\q:\qPine tree\q,\qtype\q:\q3D model\q},{\qpage\q:\qAssets\q,\qname\q:\qRock\q},{\qpage\q:\qCharacters\q,\qname\q:\qCharacter 1\q},{\qpage\q:\qLevels\q,\qname\q:\qCave\q}]}"
said "3 of 4 done. 1 was refused"
said "3. Refused: name: Characters already has a card called Character 1"
said "1. Added the asset Pine tree"
said "4. Added the level Cave"
# A wrong argument in any of them changes nothing at all, and says which one and why.
mcpjsonrefused "Bulk project" add_cards "{\qcards\q:[{\qpage\q:\qAssets\q,\qname\q:\qBush\q},{\qpage\q:\qAssets\q,\qtitle\q:\qFlower\q}]}"
said "cards number 2: add_card has no argument called title"
said "Nothing was changed."
mcp "Bulk project" search text "Bush"
said "Nothing has"
mcpjsonrefused "Bulk project" add_tasks "{\qtasks\q:\qBlock out the cave\q}"
said "tasks is a list, of up to 200, each with add_task's arguments"
# Many tasks at once, in their order.
mcpjson "Bulk project" add_tasks "{\qtasks\q:[{\qtitle\q:\qBlock out the cave\q,\qmilestone\q:\qMilestone 2\q},{\qtitle\q:\qLight the cave\q,\qmilestone\q:\qMilestone 2\q},{\qtitle\q:\qSound in the cave\q,\qmilestone\q:\qMilestone 2\q}]}"
said "All 3 done."
keep BLOCK "Block out the cave \(task (\w+)\)"
keep LIGHT "Light the cave \(task (\w+)\)"
keep SOUND "Sound in the cave \(task (\w+)\)"
# Ticked and put in order in one call: Sound first, then Light.
mcpjson "Bulk project" update_tasks "{\qtasks\q:[{\qid\q:\q$SOUND\q,\qposition\q:1},{\qid\q:\q$LIGHT\q,\qposition\q:2},{\qid\q:\q$BLOCK\q,\qdone\q:true}]}"
said "All 3 done."
# Only Milestone 2's open tasks, with titles only.
mcp "Bulk project" list_tasks milestone "Milestone 2" status "open" fields "title"
said "- Sound in the cave (task $SOUND)\n- Light the cave (task $LIGHT)\n- Task 2 (task"
said "- Task 6 (task"
notsaid "Block out the cave"
notsaid "Task 4"
notsaid "Milestone 1"
notsaid "[ ]"
mcp "Bulk project" list_tasks status "done" fields "title, done"
said "[x] Block out the cave"
said "[x] Task 1"
notsaid "Light the cave"
mcp "Bulk project" add_task title "Tidy the folder"
mcp "Bulk project" list_tasks milestone "" fields "title"
said "## Tasks with no milestone\n- Tidy the folder (task"
notsaid "Milestone 2"
mcprefused "Bulk project" list_tasks fields "title, colour"
said "fields: there's no field called colour"
mcprefused "Bulk project" list_tasks status "finished"
said "list_tasks's status is one of open or done or all; it was finished."
# Evidence: a picture is kept and can be looked at, another file is kept by its path.
writepicture "Bulk project" "shots/cave.png"
writefile "Bulk project" "logs/cave.log" "Cave built\n"
mcpjson "Bulk project" update_task "{\qid\q:\q$LIGHT\q,\qevidence\q:[\qshots/cave.png\q,\qlogs/cave.log\q]}"
said "Evidence: picture:"
said "(from shots/cave.png), the file logs/cave.log"
keep PICTURE "picture:(\S+) \(from"
mcp "Bulk project" read_picture id "picture:$PICTURE"
mcprefused "Bulk project" update_task id "$LIGHT" evidence "shots/none.png"
said "evidence: there's no file at shots/none.png in the project's folder"
mcp "Bulk project" list_tasks milestone "Milestone 2" fields "title, evidence"
said "  Evidence: picture:$PICTURE (from shots/cave.png), the file logs/cave.log"
# A milestone's notes, replaced or added to.
mcp "Bulk project" update_milestone milestone "Milestone 2" notes "The cave is lit from the entrance only."
said "Changed the notes of Milestone 2."
mcp "Bulk project" list_tasks milestone "Milestone 2" fields "notes"
said "Notes: The cave is lit from the entrance only."
mcprefused "Bulk project" update_milestone milestone "Milestone 2"
said "update_milestone needs notes"

expectnot "cave.log"

click "Edit Milestone 2"
expect "Notes"
click "Save"

said "\qPath\q:\qshots/cave.png\q"
notsaid "logs/cave.log"

# Many cards removed at once.

said "All 2 done."
mcp "Bulk project" search text "Pine"
