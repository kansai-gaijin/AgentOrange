# Made-up runs, as checks have no server; the repo, with two commits, is in the data folder.

waitgone "the default branch, at"
# A failure at the older commit shows, with its run.
testrunadd "Branch" failed "" "HEAD~1"

wait "This stays until a later run passes there."
shot branch-failed
# The newer commit passing takes it away, and the older one failing again later doesn't bring it back.
testrunadd "Branch" passed "" "HEAD"
waitgone "the default branch, at"

waitgone "the default branch, at"
